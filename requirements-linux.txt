# mismatch).

# The CPU build of onnxruntime is the baseline; the GPU builds share the
# module name, so uninstall the others first:
#   pip uninstall -y onnxruntime onnxruntime-gpu onnxruntime-directml
#   pip install onnxruntime-gpu      # CUDA EP, needs CUDA and cuDNN 9
#   pip install onnxruntime-directml # DirectML EP, Windows
